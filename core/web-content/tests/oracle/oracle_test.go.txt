package webcontent

// Differential oracle for core/web-content. Never compiled from desktop/:
// tests/oracle/regen.sh mounts it into this package with `go test -overlay`.

import (
	"encoding/json"
	"os"
	"testing"
)

type oracleResult struct {
	HTML  string `json:"html"`
	Text  string `json:"text"`
	Error string `json:"error"`
}

func oracleRun(html, text string, err error) oracleResult {
	if err != nil {
		return oracleResult{Error: err.Error()}
	}
	return oracleResult{HTML: html, Text: text}
}

func TestWriteOracle(t *testing.T) {
	in, out := os.Getenv("ORACLE_CASES"), os.Getenv("ORACLE_OUT")
	if in == "" || out == "" {
		t.Skip("oracle only")
	}
	raw, err := os.ReadFile(in)
	if err != nil {
		t.Fatal(err)
	}
	var cases struct {
		HTML []struct {
			Name  string `json:"name"`
			HTML  string `json:"html"`
			Limit int    `json:"limit"`
		} `json:"html"`
		URLs []string `json:"urls"`
	}
	if err := json.Unmarshal(raw, &cases); err != nil {
		t.Fatal(err)
	}
	type htmlOut struct {
		Name   string       `json:"name"`
		Clean  oracleResult `json:"clean"`
		Export oracleResult `json:"export"`
	}
	type urlOut struct {
		URL    string `json:"url"`
		Result string `json:"result"`
		Error  string `json:"error"`
	}
	var result struct {
		HTML []htmlOut `json:"html"`
		URLs []urlOut  `json:"urls"`
	}
	for _, c := range cases.HTML {
		limit := c.Limit
		if limit == 0 {
			limit = 64 << 20
		}
		o := htmlOut{Name: c.Name}
		o.Clean = oracleRun(Clean(c.HTML))
		o.Export = oracleRun(CleanExport(c.HTML, limit))
		result.HTML = append(result.HTML, o)
	}
	for _, u := range cases.URLs {
		got, err := SourceURL(u)
		o := urlOut{URL: u, Result: got}
		if err != nil {
			o.Error = err.Error()
		}
		result.URLs = append(result.URLs, o)
	}
	encoded, err := json.MarshalIndent(result, "", "  ")
	if err != nil {
		t.Fatal(err)
	}
	if err := os.WriteFile(out, append(encoded, '\n'), 0o644); err != nil {
		t.Fatal(err)
	}
}
